``` py

from pathlib import Path
from datetime import datetime
import re

NOTES_DIR = Path("/path/to/notes")
OUTPUT_FILE = NOTES_DIR / "_Action Tracker.md"

action_pattern = re.compile(
    r"^\s*-\s*action\s*:\s*(?P<action>.+?)\s*$",
    re.IGNORECASE,
)

status_pattern = re.compile(
    r"\[(?P<status>open|done|blocked|waiting|cancelled)\]",
    re.IGNORECASE,
)

due_pattern = re.compile(
    r"\[due:\s*(?P<due>[^\]]+)\]",
    re.IGNORECASE,
)

rows = []

for file_path in NOTES_DIR.rglob("*.md"):
    if file_path.name == OUTPUT_FILE.name:
        continue

    text = file_path.read_text(encoding="utf-8")

    for line_number, line in enumerate(text.splitlines(), start=1):
        match = action_pattern.match(line)

        if not match:
            continue

        action_text = match.group("action").strip()

        status_match = status_pattern.search(action_text)
        due_match = due_pattern.search(action_text)

        status = status_match.group("status").lower() if status_match else "open"
        due = due_match.group("due").strip() if due_match else ""

        # Remove tracking metadata from the displayed action
        clean_action = status_pattern.sub("", action_text)
        clean_action = due_pattern.sub("", clean_action).strip()

        rows.append({
            "file": file_path.name,
            "path": str(file_path.relative_to(NOTES_DIR)),
            "line": line_number,
            "action": clean_action,
            "status": status,
            "due": due,
        })

rows.sort(key=lambda row: (
    row["status"] != "open",
    row["due"] or "9999-12-31",
    row["file"],
))

output = [
    "# Action Tracker",
    "",
    f"_Generated: {datetime.now().strftime('%Y-%m-%d %H:%M')}_",
    "",
    "## Open Actions",
    "",
    "| Status | Due | Action | Source |",
    "|---|---|---|---|",
]

for row in rows:
    if row["status"] != "open":
        continue

    source = f"[{row['file']}](<{row['path']}>)"
    output.append(
        f"| {row['status']} | {row['due']} | "
        f"{row['action']} | {source}:{row['line']} |"
    )

output.extend([
    "",
    "## Other Actions",
    "",
    "| Status | Due | Action | Source |",
    "|---|---|---|---|",
])

for row in rows:
    if row["status"] == "open":
        continue

    source = f"[{row['file']}](<{row['path']}>)"
    output.append(
        f"| {row['status']} | {row['due']} | "
        f"{row['action']} | {source}:{row['line']} |"
    )

OUTPUT_FILE.write_text("\n".join(output) + "\n", encoding="utf-8")

print(f"Wrote {len(rows)} actions to {OUTPUT_FILE}")


```
